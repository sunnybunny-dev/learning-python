
#Python Sets
Definition: Unordered collection of unique elements.
Create: my_set ={1,2,3} or my_set = set([1,2,3])
Add element: my_set.add(4)
Remove element: my_set.remove(2)

#Python Lists
def: Ordeered,mutable and collection of elements
Create: my_list =[1,2,3]
Add element: my_list.append(4)
Remove element: my_list.remove(2)

#Python Dictionaries
Definition: Unordered collection of key-value pairs.
Create: my_dict = {'a': 1, 'b': 2}
Add/Update: my_dict['c'] = 3
Remove: del my_dict['a']
